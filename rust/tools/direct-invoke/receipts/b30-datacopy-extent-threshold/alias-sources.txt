#include "kernel_operator.h"

#ifndef TILE_OUT_FILL
#define TILE_OUT_FILL 0
#endif

class Kernel_v41_embedding_5120 {
public:
    __aicore__ inline Kernel_v41_embedding_5120() {}
    __aicore__ inline void Init(GM_ADDR gm0, GM_ADDR gm1, GM_ADDR gm2) {
        gm0_raw = (uint64_t)(uintptr_t)gm0;
        gm1_raw = (uint64_t)(uintptr_t)gm1;
        gm2_raw = (uint64_t)(uintptr_t)gm2;
        gm0Tensor.SetGlobalBuffer((__gm__ float*)gm0, 16777216);
        gm1Tensor.SetGlobalBuffer((__gm__ float*)gm1, 16777216);
        gm2Tensor.SetGlobalBuffer((__gm__ float*)gm2, 16777216);
        pipe.InitBuffer(tbuf0, 32);
        pipe.InitBuffer(tbuf1, 20480);
    }
    __aicore__ inline void Process() {
        AscendC::LocalTensor<float> buf0 = tbuf0.Get<float>();
        AscendC::LocalTensor<float> buf1 = tbuf1.Get<float>();
        int32_t __v0;
        int32_t __v1;
        int32_t __v2;
        int32_t __v3;
        float __vidx;
        int32_t __v4;
        __v0 = 1;
        __v1 = 5120;
        __v2 = 1;
        __v3 = 8;
        AscendC::Duplicate(buf0, 2.0f, 8);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::DataCopyExtParams __lp{1, 4, 0, 0, 0};
        AscendC::DataCopyPadExtParams<float> __lq{false, 0, 0, 0};
        AscendC::DataCopyPad(buf0, gm1Tensor, __lp, __lq);
        AscendC::PipeBarrier<PIPE_ALL>();
        buf0.SetValue(0, 1.0f);
        AscendC::PipeBarrier<PIPE_ALL>();
        __vidx = (float)buf0.GetValue((int32_t)(__v0));
        __v4 = __vidx * __v1;
        AscendC::DataCopy(buf1, gm0Tensor[__v4], (__v2) * (__v1));
                                       AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::PipeBarrier<PIPE_ALL>();
        { const uint32_t __sb = (uint32_t)(((__v2) * (__v1)) * sizeof(float));
          if (__sb % 32 == 0) { AscendC::DataCopy(gm2Tensor, buf1, ((__v2) * (__v1))); }
          else { AscendC::DataCopyExtParams __sp{1, __sb, 0, 0, 0};
                 AscendC::DataCopyPad(gm2Tensor, buf1, __sp); } }
        return;
    }
private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tbuf0;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tbuf1;
    AscendC::GlobalTensor<float> gm0Tensor;
    uint64_t gm0_raw;
    AscendC::GlobalTensor<float> gm1Tensor;
    uint64_t gm1_raw;
    AscendC::GlobalTensor<float> gm2Tensor;
    uint64_t gm2_raw;
};

extern "C" __global__ __aicore__ void v41_embedding_5120(GM_ADDR gm0, GM_ADDR gm1, GM_ADDR gm2) {
    Kernel_v41_embedding_5120 op;
    op.Init(gm0, gm1, gm2);
    op.Process();
}


extern "C" void v41_embedding_5120_do(uint32_t blockDim, void *stream, uint8_t *gm0, uint8_t *gm1, uint8_t *gm2) {
    v41_embedding_5120<<<blockDim, nullptr, stream>>>(gm0, gm1, gm2);
}
